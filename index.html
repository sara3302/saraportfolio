# celery_tasks.py — Asynchronous Task Queue for Medi Backend 
from celery import shared_task
from .ml_models import predict_heart_disease
from .report_parser import parse_clinical_report

@shared_task(bind=True, max_retries=3)
def process_patient_lab_report(self, report_id: int, raw_text: str):
    """
    Dispatches asynchronous diagnostic parsing via OpenAI API 
    and updates database record in PostgreSQL.
    """
    try:
        parsed_data = parse_clinical_report(raw_text)
        risk_score = predict_heart_disease(parsed_data)
        
        # Atomically update DB record
        update_lab_record(report_id, parsed_data, risk_score)
        return {"status": "SUCCESS", "report_id": report_id, "risk": risk_score}
    except Exception as exc:
        self.retry(exc=exc, countdown=10)
# report_parser.py — Structured JSON Extraction via OpenAI API 
import openai
import json

def parse_clinical_report(raw_report_text: str) -> dict:
    """
    Parses unstructured clinical report text into a validated JSON schema
    using OpenAI API structured outputs.
    """
    client = openai.OpenAI()
    system_prompt = (
        "You are a medical data extraction engine. Extract clinical markers "
        "and return ONLY valid JSON with keys: cholesterol, systolic_bp, "
        "chest_pain_type, and glucose."
    )
    
    response = client.chat.completions.create(
        model="gpt-4o-mini",
        response_format={"type": "json_object"},
        messages=[
            {"role": "system", "content": system_prompt},
            {"role": "user", "content": raw_report_text}
        ],
        temperature=0.1
    )
    return json.loads(response.choices[0].message.content)
