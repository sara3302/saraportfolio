# celery_tasks.py — Asynchronous Task Queue for Medi Backend from celery import shared_task from .ml_models import predict_heart_disease from .report_parser import parse_clinical_report @shared_task(bind=True, max_retries=3) def process_patient_lab_report(self, report_id: int, raw_text: str): """ Dispatches asynchronous diagnostic parsing via Gemini API and updates database record in PostgreSQL. """ try: parsed_data = parse_clinical_report(raw_text) risk_score = predict_heart_disease(parsed_data) # Atomically update DB record update_lab_record(report_id, parsed_data, risk_score) return {"status": "SUCCESS", "report_id": report_id, "risk": risk_score} except Exception as exc: self.retry(exc=exc, countdown=10)
# report_parser.py — Structured JSON Extraction via Gemini API import google.generativeai as genai import json def parse_clinical_report(raw_report_text: str) -> dict: model = genai.GenerativeModel('gemini-1.5-flash') prompt = f""" Extract structured medical diagnostic markers from the input clinical report. Return ONLY valid JSON matching this schema: {{ "cholesterol": float, "systolic_bp": int, "chest_pain_type": int, "glucose": float }} Input Report: {raw_report_text} """ response = model.generate_content(prompt) return json.loads(response.text)
